 27    A class used to represent an FSM.

 43    """

 45    name: str
 46    states: dict[str, State]
 47    inputs: list[str] = field(init=False)
 48    outputs: list[str] = field(init=False)
 49    start_state: str = field(init=False)
 50    fsm_type: FSMType = field(init=False)
 51
 52    def __post_init__(self) -> None:
 53        """
 54        Executes after the name and states fields have been populated and
 55        computes derived attributes that depend on the state dictionary.
 56        """
 57
 58        if determine_missing_source_state(self.states) is not None:
 59            add_missing_source_states(self.states)
 60
 61        self.inputs, self.outputs = identify_unique_io(self.states)
 62
 63        # self.start_state: str = sorted(self.states.keys())[0]
 64        self.start_state: str = sorted(
 65            self.states.keys(), key=lambda x: int(x[1 : len(x.split("_")[0])])
 66        )[0]
 67
 68        self.fsm_type = self.identify_fsm_type()
 69
 70        # if Moore then set outputs for each state object
 71        if self.fsm_type == FSMType.MOORE:
 72            set_fsm_outputs_moore(self.states)
 73
 74    def identify_fsm_type(self) -> FSMType:
 75        """
 76        Identifies the behavioral model (or type) of the FSM.
 77
 78        Returns
 79        -------
 80        ``FSMType``
 81            The behavioral model (or type) of the FSM.
 82        """
 83
 84        # loop through all state names
 85        state_name: str
 86        for state_name in self.states:
 87            # keep first transition
 88            first_transition_outputs: list[str]
 89            is_first_transition: bool = True
 90
 91            # loop through all states
 92            state: State
 93            for state in self.states.values():
 94                # loop through all transitions
 95                transition: Transition
 96                for transition in state.transitions:
 97                    # target is the current state which is being checked
 98                    if transition.state_target == state_name:
 99                        # store first transition outputs
100                        if is_first_transition:
101                            first_transition_outputs = transition.outputs
102                            is_first_transition = False
103                            continue
104                        # check if the transition outputs are the same
105                        if not check_state_transition_outputs_same(
106                            transition.outputs, first_transition_outputs
107                        ):
108                            return FSMType.MEALY
109
110        # if all checks pass the FSM is of type Moore
111        return FSMType.MOORE
